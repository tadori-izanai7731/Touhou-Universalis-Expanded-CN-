THE_SEJ_downfall_revolution_refresh_border_opinion_effect = {
	every_country = {
		limit = {
			THE_SEJ_downfall_revolution_foreign_target_st = yes
			NOT = { is_neighbor_of = ROOT }
			has_opinion_modifier = {
				who = ROOT
				modifier = THE_SEJ_downfall_revolution_pressure_opinion
			}
		}
		remove_opinion = {
			who = ROOT
			modifier = THE_SEJ_downfall_revolution_pressure_opinion
		}
	}
	every_neighbor_country = {
		limit = {
			THE_SEJ_downfall_revolution_foreign_target_st = yes
			NOT = {
				has_opinion_modifier = {
					who = ROOT
					modifier = THE_SEJ_downfall_revolution_pressure_opinion
				}
			}
		}
		add_opinion = {
			who = ROOT
			modifier = THE_SEJ_downfall_revolution_pressure_opinion
		}
	}
}
